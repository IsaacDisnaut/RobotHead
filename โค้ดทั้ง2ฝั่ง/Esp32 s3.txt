uint8_t clampThreshold(int16_t value)
{
  if (value < 0) return 0;
  if (value > 254) return 254;
  return static_cast<uint8_t>(value);
}

bool beginPanels()
{
  pinMode(Pins::kLeftCs, OUTPUT);
  pinMode(Pins::kRightCs, OUTPUT);
  digitalWrite(Pins::kLeftCs, HIGH);
  digitalWrite(Pins::kRightCs, HIGH);

  sharedBus = new Arduino_ESP32SPI(Pins::kDc, GFX_NOT_DEFINED, Pins::kSclk, Pins::kDin);
  if (!sharedBus) return false;

  leftPanel = new Arduino_GC9A01(sharedBus, Pins::kLeftReset, DisplayConfig::kLeftRotation, true /* IPS */);
  rightPanel = new Arduino_GC9A01(sharedBus, Pins::kRightReset, DisplayConfig::kRightRotation, true /* IPS */);
  if (!leftPanel || !rightPanel) return false;

  digitalWrite(Pins::kRightCs, HIGH);
  digitalWrite(Pins::kLeftCs, LOW);
  if (!leftPanel->begin(DisplayConfig::kSpiFrequency)) return false;
  leftPanel->fillScreen(DisplayConfig::kPanelBackground);
  digitalWrite(Pins::kLeftCs, HIGH);
  delay(100);

  digitalWrite(Pins::kLeftCs, HIGH);
  digitalWrite(Pins::kRightCs, LOW);
  if (!rightPanel->begin(DisplayConfig::kSpiFrequency)) return false;
  rightPanel->fillScreen(DisplayConfig::kPanelBackground);
  digitalWrite(Pins::kRightCs, HIGH);
  delay(100);

  eyes[0].panel = leftPanel;
  eyes[1].panel = rightPanel;
  return true;
}

void drawEye(
    uint8_t eyeIndex,
    uint32_t iScale,
    uint32_t scleraX,
    uint32_t scleraY,
    uint32_t upperThreshold,
    uint32_t lowerThreshold)
{
  if (!eyes[eyeIndex].panel) return;

  uint32_t screenX, screenY;
  uint32_t scleraXStart = scleraX;
  int32_t irisXStart = static_cast<int32_t>(scleraXStart) - ((SCLERA_WIDTH - IRIS_WIDTH) / 2);
  int32_t irisYStart = static_cast<int32_t>(scleraY) - ((SCLERA_HEIGHT - IRIS_HEIGHT) / 2);
  size_t pixels = 0;

  if (eyeIndex == 0) {
    digitalWrite(Pins::kRightCs, HIGH);
    digitalWrite(Pins::kLeftCs, LOW);
  } else {
    digitalWrite(Pins::kLeftCs, HIGH);
    digitalWrite(Pins::kRightCs, LOW);
  }

  Arduino_GC9A01 *panel = eyes[eyeIndex].panel;
  panel->startWrite();
  panel->writeAddrWindow(kRenderXOffset, kRenderYOffset, kRenderWidth, kRenderHeight);

  for (screenY = 0; screenY < kRenderHeight; ++screenY) {
    uint32_t sourceScreenY = (screenY * SCREEN_HEIGHT) / kRenderHeight;
    uint32_t sourceScleraY = scleraY + sourceScreenY;
    int32_t sourceIrisY = irisYStart + static_cast<int32_t>(sourceScreenY);

    for (screenX = 0; screenX < kRenderWidth; ++screenX) {
      uint16_t pixel;
      uint32_t sourceScreenX = (screenX * SCREEN_WIDTH) / kRenderWidth;
      uint32_t sourceScleraX = scleraXStart + sourceScreenX;
      int32_t sourceIrisX = irisXStart + static_cast<int32_t>(sourceScreenX);
      uint16_t lidX = (eyeIndex == 0) ? (SCREEN_WIDTH - 1 - sourceScreenX) : sourceScreenX;

      if (sourceScreenY >= SCREEN_HEIGHT || sourceScreenX >= SCREEN_WIDTH ||
          sourceScleraY >= SCLERA_HEIGHT || sourceScleraX >= SCLERA_WIDTH) {
        pixel = 0x0000;
      }
      else if ((pgm_read_byte(lower + sourceScreenY * SCREEN_WIDTH + lidX) <= lowerThreshold) ||
               (pgm_read_byte(upper + sourceScreenY * SCREEN_WIDTH + lidX) <= upperThreshold)) {
        pixel = 0x0000;
      }
      else if ((sourceIrisY < 0) || (sourceIrisY >= IRIS_HEIGHT) || 
               (sourceIrisX < 0) || (sourceIrisX >= IRIS_WIDTH)) {
        pixel = pgm_read_word(sclera + sourceScleraY * SCLERA_WIDTH + sourceScleraX);
      }
      else {
        uint16_t polarValue = pgm_read_word(polar + sourceIrisY * IRIS_WIDTH + sourceIrisX);
        uint32_t distance = (iScale * (polarValue & 0x7F)) / 128;
        if (distance < IRIS_MAP_HEIGHT) {
          uint32_t angle = (IRIS_MAP_WIDTH * (polarValue >> 7)) / 512;
          pixel = pgm_read_word(iris + distance * IRIS_MAP_WIDTH + angle);
        } else {
          pixel = pgm_read_word(sclera + sourceScleraY * SCLERA_WIDTH + sourceScleraX);
        }
      }

      pixelBuffer[pixels++] = pixel;
      if (pixels >= PIXEL_BUFFER_SIZE) {
        panel->writePixels(pixelBuffer, pixels);
        pixels = 0;
      }
    }
  }

  if (pixels != 0) {
    panel->writePixels(pixelBuffer, pixels);
  }

  panel->endWrite();
  digitalWrite(Pins::kLeftCs, HIGH);
  digitalWrite(Pins::kRightCs, HIGH);
}

void checkSerialCommand()
{
  static String rxBuffer = "";
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n') {
      rxBuffer.trim();
      if (rxBuffer.startsWith("T,")) {
        int c1 = rxBuffer.indexOf(',');
        int c2 = rxBuffer.indexOf(',', c1 + 1);
        int c3 = rxBuffer.indexOf(',', c2 + 1);
        int c4 = rxBuffer.indexOf(',', c3 + 1);
        if (c1 > 0 && c2 > 0 && c3 > 0 && c4 > 0) {
          targetPan = rxBuffer.substring(c1 + 1, c2).toFloat();
          targetTilt = rxBuffer.substring(c2 + 1, c3).toFloat();
          pcEyeX = rxBuffer.substring(c3 + 1, c4).toInt();
          pcEyeY = rxBuffer.substring(c4 + 1).toInt();
          pcTrackingMode = true;
          lastPcMessageTime = millis();
        }
      } else if (rxBuffer == "NOD") {
        isNodding = true;
        nodStartTime = millis();
      } else if (rxBuffer == "TILT") {
        // ⭐ คำสั่งเอียงคอรับฟังอย่างตั้งใจ
        isTilting = true;
        tiltStartTime = millis();
      } else if (rxBuffer == "THINK") {
        pcEyeX = 512;
        pcEyeY = 250; // เหลือบตามองบนนึกคิด
      } else if (rxBuffer.startsWith("I")) {
        pcTrackingMode = false;
        targetPan = 90.0f;
        targetTilt = 90.0f;
      }
      rxBuffer = "";
    } else {
      if (rxBuffer.length() < 64) rxBuffer += c;
      else rxBuffer = "";
    }
  }

  if (pcTrackingMode && (millis() - lastPcMessageTime > 2500)) {
    pcTrackingMode = false;
    targetPan = 90.0f;
    targetTilt = 90.0f;
  }
}

void frame(uint16_t iScale)
{
  static uint8_t eyeIndex = NUM_EYES - 1;
  static bool eyeInMotion = false;
  static int16_t eyeOldX = 512, eyeOldY = 512;
  static int16_t eyeNewX = 512, eyeNewY = 512;
  static uint32_t eyeMoveStartTime = 0;
  static int32_t eyeMoveDuration = 0;

  int16_t eyeX, eyeY;
  const uint32_t now = micros();

  checkSerialCommand();

  float effectiveTilt = targetTilt;
  float effectivePan  = targetPan;

  // แอนิเมชันภาษากาย
  if (isNodding) {
    uint32_t nodElapsed = millis() - nodStartTime;
    if (nodElapsed < 300) {
      effectiveTilt += 15.0f;
    } else if (nodElapsed < 600) {
      effectiveTilt -= 5.0f;
    } else {
      isNodding = false;
    }
  } else if (isTilting) {
    uint32_t tiltElapsed = millis() - tiltStartTime;
    if (tiltElapsed < 800) {
      effectivePan += 6.0f;  // เอียงคอเล็กน้อย
      effectiveTilt += 4.0f;
    } else {
      isTilting = false;
    }
  }

  currentPan  += (effectivePan - currentPan) * 0.14f;
  currentTilt += (effectiveTilt - currentTilt) * 0.14f;
  servoPan.write(constrain(static_cast<int>(currentPan), 30, 150));
  servoTilt.write(constrain(static_cast<int>(currentTilt), 50, 130));

  if (++eyeIndex >= NUM_EYES) {
    eyeIndex = 0;
  }

  if (pcTrackingMode) {
    eyeX = pcEyeX;
    eyeY = pcEyeY;
  } else {
    static uint32_t lastIdleHeadMove = 0;
    if (millis() - lastIdleHeadMove > 4000) {
      lastIdleHeadMove = millis();
      targetPan = 90.0f + random(-10, 11);
      targetTilt = 90.0f + random(-5, 6);
    }

    int32_t dt = static_cast<int32_t>(now - eyeMoveStartTime);
    if (eyeInMotion) {
      if (dt >= eyeMoveDuration) {
        eyeInMotion = false;
        eyeMoveDuration = random(kActiveMood.movePauseMinUs, kActiveMood.movePauseMaxUs);
        eyeMoveStartTime = now;
        eyeX = eyeOldX = eyeNewX;
        eyeY = eyeOldY = eyeNewY;
      } else {
        int16_t easeIndex = (255 * dt) / eyeMoveDuration;
        if (easeIndex < 0) easeIndex = 0;
        else if (easeIndex > 255) easeIndex = 255;
        int16_t e = pgm_read_byte(ease + easeIndex) + 1;
        eyeX = eyeOldX + (((eyeNewX - eyeOldX) * e) / 256);
        eyeY = eyeOldY + (((eyeNewY - eyeOldY) * e) / 256);
      }
    } else {
      eyeX = eyeOldX;
      eyeY = eyeOldY;
      if (dt > eyeMoveDuration) {
        int16_t dx, dy;
        uint32_t distance;
        do {
          eyeNewX = random(1024);
          eyeNewY = random(1024);
          dx = (eyeNewX * 2) - 1023;
          dy = (eyeNewY * 2) - 1023;
          distance = static_cast<uint32_t>(dx * dx + dy * dy);
        } while (distance > (1023UL * 1023UL));
        eyeMoveDuration = random(kActiveMood.moveTimeMinUs, kActiveMood.moveTimeMaxUs);
        eyeMoveStartTime = now;
        eyeInMotion = true;
      }
    }
  }

  if ((now - timeOfLastBlink) >= timeToNextBlink) {
    timeOfLastBlink = now;
    uint32_t blinkDuration = random(kActiveMood.blinkMinUs, kActiveMood.blinkMaxUs);
    for (uint8_t e = 0; e < NUM_EYES; ++e) {
      if (eyes[e].blink.state == NOBLINK) {
        eyes[e].blink.state = ENBLINK;
        eyes[e].blink.startTime = now;
        eyes[e].blink.duration = blinkDuration;
      }
    }
    timeToNextBlink = random(kActiveMood.blinkGapMinUs, kActiveMood.blinkGapMaxUs);
  }

  if (eyes[eyeIndex].blink.state != NOBLINK) {
    if ((now - eyes[eyeIndex].blink.startTime) >= eyes[eyeIndex].blink.duration) {
      if (eyes[eyeIndex].blink.state == ENBLINK) {
        eyes[eyeIndex].blink.state = DEBLINK;
        eyes[eyeIndex].blink.duration *= 2;
        eyes[eyeIndex].blink.startTime = now;
      } else {
        eyes[eyeIndex].blink.state = NOBLINK;
      }
    }
  }

  eyeX = map(eyeX, 0, 1023, 0, SCLERA_WIDTH - SCREEN_WIDTH);
  eyeY = map(eyeY, 0, 1023, 0, SCLERA_HEIGHT - SCREEN_HEIGHT);
  if (eyeX < 0) eyeX = 0;
  else if (eyeX > (SCLERA_WIDTH - SCREEN_WIDTH)) eyeX = (SCLERA_WIDTH - SCREEN_WIDTH);

  static uint8_t upperThreshold = 128;
  int16_t sampleX = (SCLERA_WIDTH / 2) - (eyeX / 2);
  int16_t sampleY = (SCLERA_HEIGHT / 2) - (eyeY + (IRIS_HEIGHT / 4));
  uint8_t eyelidSample = (sampleY < 0) ? 0 : (pgm_read_byte(upper + sampleY * SCREEN_WIDTH + sampleX) +
                         pgm_read_byte(upper + sampleY * SCREEN_WIDTH + (SCREEN_WIDTH - 1 - sampleX))) / 2;
  upperThreshold = (upperThreshold * 3 + eyelidSample) / 4;
  uint8_t currentUpperThreshold = clampThreshold(static_cast<int16_t>(upperThreshold) + kActiveMood.upperLidOffset);
  uint8_t lowerThreshold = clampThreshold(static_cast<int16_t>(254 - upperThreshold) + kActiveMood.lowerLidOffset);

  if (eyes[eyeIndex].blink.state != NOBLINK) {
    uint32_t blinkProgress = now - eyes[eyeIndex].blink.startTime;
    blinkProgress = (blinkProgress >= eyes[eyeIndex].blink.duration) ? 255 : (255 * blinkProgress) / eyes[eyeIndex].blink.duration;
    blinkProgress = (eyes[eyeIndex].blink.state == DEBLINK) ? (1 + blinkProgress) : (256 - blinkProgress);
    currentUpperThreshold = (currentUpperThreshold * blinkProgress + 254 * (257 - blinkProgress)) / 256;
    lowerThreshold = (lowerThreshold * blinkProgress + 254 * (257 - blinkProgress)) / 256;
  }

  drawEye(eyeIndex, iScale, eyeX, eyeY, currentUpperThreshold, lowerThreshold);
}

void split(int16_t startValue, int16_t endValue, uint32_t startTime, int32_t duration, int16_t range)
{
  if (range >= 8) {
    range /= 2; duration /= 2;
    int16_t midValue = (startValue + endValue - range) / 2 + random(range);
    split(startValue, midValue, startTime, duration, range);
    split(midValue, endValue, startTime + duration, duration, range);
  } else {
    int32_t dt;
    while ((dt = static_cast<int32_t>(micros() - startTime)) < duration) {
      int16_t value = startValue + (((endValue - startValue) * dt) / duration);
      frame(value);
    }
  }
}

void updateEye()
{
  newIris = random(95, 125);
  split(oldIris, newIris, micros(), kActiveMood.irisTransitionUs, 30);
  oldIris = newIris;
}

void setup()
{
  Serial.begin(SERIAL_BAUD);
  delay(500);

  ESP32PWM::allocateTimer(0);
  ESP32PWM::allocateTimer(1);
  servoPan.setPeriodHertz(50);
  servoTilt.setPeriodHertz(50);
  servoPan.attach(Pins::kServoPan, 500, 2400);
  servoTilt.attach(Pins::kServoTilt, 500, 2400);
  servoPan.write(90);
  servoTilt.write(90);

  randomSeed(micros());
  for (uint8_t e = 0; e < NUM_EYES; ++e) {
    eyes[e].blink.state = NOBLINK;
  }

  if (!beginPanels()) {
    while (true) delay(1000);
  }

  timeOfLastBlink = micros();
  timeToNextBlink = random(kActiveMood.blinkGapMinUs, kActiveMood.blinkGapMaxUs);
}

void loop()
{
  updateEye();
}
